`timescale 1ns/1ps
// Educational RV32I subset, five-stage pipeline. Word-aligned LW/SW only.
module cpu #(parameter IMEM_WORDS=256, DMEM_WORDS=256)(
 input wire clk, input wire reset,
 output wire [31:0] debug_pc,
 output wire stall, output wire flush
);
 localparam NOP=32'h00000013;
 reg [31:0] pc;
 reg [31:0] imem [0:IMEM_WORDS-1];
 reg [31:0] dmem [0:DMEM_WORDS-1];
 reg [31:0] regs [0:31];
 integer i;
 initial begin
   for (i=0;i<IMEM_WORDS;i=i+1) imem[i]=NOP;
   for (i=0;i<DMEM_WORDS;i=i+1) dmem[i]=0;
   for (i=0;i<32;i=i+1) regs[i]=0;
 end
 assign debug_pc=pc;
 wire [31:0] fetched = (pc[31:2]<IMEM_WORDS) ? imem[pc[31:2]] : NOP;
 // IF/ID
 reg if_valid;
 reg [31:0] if_pc, if_insn;
 wire [6:0] op=if_insn[6:0];
 wire [4:0] rs1=if_insn[19:15],rs2=if_insn[24:20],rd=if_insn[11:7];
 wire [2:0] funct3=if_insn[14:12];
 wire [6:0] funct7=if_insn[31:25];
 wire is_add=(op==7'b0110011 && funct3==0 && funct7==0);
 wire is_addi=(op==7'b0010011 && funct3==0);
 wire is_lw=(op==7'b0000011 && funct3==3'b010);
 wire is_sw=(op==7'b0100011 && funct3==3'b010);
 wire is_beq=(op==7'b1100011 && funct3==0);
 wire is_lui=(op==7'b0110111);
 wire is_jal=(op==7'b1101111);
 wire is_jalr=(op==7'b1100111 && funct3==0);
 wire id_supported=is_add|is_addi|is_lw|is_sw|is_beq|is_lui|is_jal|is_jalr;
 wire id_uses_rs1=is_add|is_addi|is_lw|is_sw|is_beq|is_jalr;
 wire id_uses_rs2=is_add|is_sw|is_beq;
 wire [31:0] imm_i={{20{if_insn[31]}},if_insn[31:20]};
 wire [31:0] imm_s={{20{if_insn[31]}},if_insn[31:25],if_insn[11:7]};
 wire [31:0] imm_b={{19{if_insn[31]}},if_insn[31],if_insn[7],if_insn[30:25],if_insn[11:8],1'b0};
 wire [31:0] imm_j={{11{if_insn[31]}},if_insn[31],if_insn[19:12],if_insn[20],if_insn[30:21],1'b0};
 wire [31:0] id_imm=is_sw?imm_s:is_beq?imm_b:is_jal?imm_j:is_lui?{if_insn[31:12],12'b0}:imm_i;
 // ID/EX
 reg ex_valid;
 reg [31:0] ex_pc,ex_a,ex_b,ex_imm;
 reg [4:0] ex_rs1,ex_rs2,ex_rd;
 reg ex_add,ex_addi,ex_lw,ex_sw,ex_beq,ex_lui,ex_jal,ex_jalr;
 // EX/MEM
 reg mem_valid,mem_regwrite,mem_load,mem_store;
 reg [31:0] mem_result,mem_store_data,mem_addr;
 reg [4:0] mem_rd,mem_store_rs2;
 // MEM/WB
 reg wb_valid,wb_regwrite;
 reg [4:0] wb_rd;
 reg [31:0] wb_data;
 wire mem_can_forward=mem_valid && mem_regwrite && !mem_load && mem_rd!=0;
 wire wb_can_forward=wb_valid && wb_regwrite && wb_rd!=0;
 wire [31:0] fwd_a=(mem_can_forward && mem_rd==ex_rs1)?mem_result:
                    (wb_can_forward && wb_rd==ex_rs1)?wb_data:ex_a;
 wire [31:0] fwd_b=(mem_can_forward && mem_rd==ex_rs2)?mem_result:
                    (wb_can_forward && wb_rd==ex_rs2)?wb_data:ex_b;
 wire [31:0] ex_result=ex_lui?ex_imm:(ex_jal||ex_jalr)?(ex_pc+4):
                       fwd_a+(ex_addi||ex_lw||ex_sw?ex_imm:fwd_b);
 wire redirect=ex_valid && (ex_jal || ex_jalr || (ex_beq && fwd_a==fwd_b));
 wire [31:0] target=ex_jalr?((fwd_a+ex_imm)&32'hfffffffe):ex_pc+ex_imm;
 assign flush=redirect;
 assign stall=if_valid && ex_valid && ex_lw && ex_rd!=0 &&
             ((id_uses_rs1 && rs1==ex_rd)||(id_uses_rs2 && rs2==ex_rd));
 // Store-data late forwarding for LW->SW (store value can arrive at MEM).
 wire [31:0] store_value=(mem_store && wb_can_forward && wb_rd==mem_store_rs2)?wb_data:mem_store_data;
 wire [31:0] load_value=(mem_addr[31:2]<DMEM_WORDS)?dmem[mem_addr[31:2]]:32'b0;
 always @(posedge clk) begin
   if (reset) begin
     pc<=0; if_valid<=0; if_pc<=0; if_insn<=NOP;
     ex_valid<=0; ex_pc<=0;ex_a<=0;ex_b<=0;ex_imm<=0;
     ex_rs1<=0;ex_rs2<=0;ex_rd<=0;
     ex_add<=0;ex_addi<=0;ex_lw<=0;ex_sw<=0;ex_beq<=0;ex_lui<=0;ex_jal<=0;ex_jalr<=0;
     mem_valid<=0;mem_regwrite<=0;mem_load<=0;mem_store<=0;
     mem_result<=0;mem_store_data<=0;mem_addr<=0;mem_rd<=0;mem_store_rs2<=0;
     wb_valid<=0;wb_regwrite<=0;wb_rd<=0;wb_data<=0;
   end else begin
     // WB: nonblocking write; ID reads include same-cycle bypass below.
     if (wb_can_forward) regs[wb_rd]<=wb_data;
     regs[0]<=0;
     // MEM
     if (mem_valid && mem_store && mem_addr[31:2]<DMEM_WORDS)
       dmem[mem_addr[31:2]]<=store_value;
     wb_valid<=mem_valid;
     wb_regwrite<=mem_regwrite;
     wb_rd<=mem_rd;
     wb_data<=mem_load?load_value:mem_result;
     // EX
     mem_valid<=ex_valid;
     mem_regwrite<=ex_add|ex_addi|ex_lw|ex_lui|ex_jal|ex_jalr;
     mem_load<=ex_lw;mem_store<=ex_sw;
     mem_rd<=ex_rd;mem_result<=ex_result;
     mem_addr<=fwd_a+ex_imm;
     mem_store_data<=fwd_b;mem_store_rs2<=ex_rs2;
     // ID -> EX: insert bubble on stall, or squash on redirect.
     ex_valid<=if_valid && id_supported && !stall && !redirect;
     ex_pc<=if_pc;ex_rs1<=rs1;ex_rs2<=rs2;ex_rd<=rd;
     ex_a<=(wb_can_forward && wb_rd==rs1)?wb_data:regs[rs1];
     ex_b<=(wb_can_forward && wb_rd==rs2)?wb_data:regs[rs2];
     ex_imm<=id_imm;
     ex_add<=is_add;ex_addi<=is_addi;ex_lw<=is_lw;ex_sw<=is_sw;
     ex_beq<=is_beq;ex_lui<=is_lui;ex_jal<=is_jal;ex_jalr<=is_jalr;
     if (redirect) begin
       pc<=target; if_valid<=0; if_insn<=NOP;
     end else if (!stall) begin
       if_valid<=1;if_pc<=pc;if_insn<=fetched;pc<=pc+4;
     end
   end
 end
endmodule
